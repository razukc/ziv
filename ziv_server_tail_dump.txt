 1100	            await _push(frame_for(turn.events[-1]))
 1101	
 1102	            # Content on the motors (the channel stays held through every
 1103	            # cell dwell — a message arriving mid-content queues,
 1104	            # invariant 4). The wrist is freed by the LAST close of the
 1105	            # sequence: the main close when the queue holds nothing AT THE
 1106	            # CLOSE PUSH, else the final replay's close below. The flag is
 1107	            # evaluated lazily (see _play_cells_and_close) — after the
 1108	            # dwells, where fills land.
 1109	            await _play_cells_and_close(text, free_fn=lambda: len(gate) == 0)
 1110	            owed.pop(0)  # this text's own close played — the wearer has it
 1111	
 1112	            # The close has played; the gate releases (invariants 2 + 4)
 1113	            # and the drained messages replay as their own full events (cue
 1114	            # → cells → close), the last one carrying the ``free`` marker
 1115	            # the dev band fires its armed redial on.
 1116	            replay = turn.finish_playback(now_s=round(time.monotonic() - t0, 1))
 1117	            drained = replay[1:]
 1118	            released = True  # the gate is the next arrival's now
 1119	            await _play_drained(drained, owed)
 1120	
 1121	            _telemetry.record(
 1122	                label="message_turn",
 1123	                seconds=turn.seconds,
 1124	                retries=turn.retries,
 1125	                source=source,
 1126	                queued=len(drained),
 1127	            )
 1128	            return {
 1129	                "event": "turn_complete",
 1130	                "seconds": turn.seconds,
 1131	                "events": [ev.event_type for ev in turn.events],
 1132	                "drained": len(drained),
 1133	            }
 1134	    except BaseException as exc:
 1135	        # The turn died part-way through (the real case: the band vanished
 1136	        # and _push raised _NoDevices; also a cancellation while queued for
 1137	        # the turn lock). Two promises hold here, and both are about what
 1138	        # the NEXT message finds:
 1139	        #
 1140	        #   1. the channel is released. gate.admit() holds it for this
 1141	        #      whole turn and only finish_playback() clears it — a turn
 1142	        #      that raises in between leaves _playing True forever, every
 1143	        #      later arrival queues against a wrist that is not reading,
 1144	        #      and after MAX_QUEUED the relay refuses everything until
 1145	        #      restart. abort() clears it.
 1146	        #   2. nothing the wearer never felt is dropped. The in-flight
 1147	        #      text and every unplayed replay go to the inbox, which
 1148	        #      delivers them as full events on the next attach. The
 1149	        #      still-queued texts come back from abort() for the same
 1150	        #      reason — the gate no longer holds them. Only while the
 1151	        #      channel is still ours: past ``released`` the queue belongs to
 1152	        #      another turn, and touching it would corrupt their state.
 1153	        if not released:
 1154	            owed.extend(gate.abort())
 1155	        for pending_text in owed:
 1156	            inbox.offer(pending_text, source=source)
 1157	        _telemetry.record(
 1158	            label="message_turn_aborted",
 1159	            seconds=0.0,
 1160	            retries=turn.retries if turn is not None else 0,
 1161	            source=source,
 1162	            reason=type(exc).__name__,
 1163	            stranded=len(owed),
 1164	        )
 1165	        LOGGER.warning(
 1166	            "turn_aborted source=%s reason=%s stranded=%d — "
 1167	            "gate released, undelivered texts stored in the inbox",
 1168	            source, type(exc).__name__, len(owed),
 1169	        )
 1170	        raise

 1178	async def ws_device(ws: WebSocket, token: str | None = Query(default=None)) -> None:
 1179	    """The device transport: attach, greet, drain, detach."""
 1180	    if not _authorized_websocket(token):
 1181	        await ws.close(code=4401)
 1182	        return
 1183	    await ws.accept()
 1184	    await hub.attach(ws)
 1185	    try:
 1186	        pending = inbox.count()
 1187	        await ws.send_text(
 1188	            json.dumps(
 1189	                {
 1190	                    "type": "hello",
 1191	                    "spec_version": timing.SPEC_VERSION,
 1192	                    "patterns": list(timing.PATTERN_IDS),
 1193	                    "mark": timing.PREFIX_MARK,
 1194	                    "inbox_pending": pending,
 1195	                },
 1196	                ensure_ascii=False,
 1197	            )
 1198	        )
 1199	        # Relay v1: a message that arrived while no band was attached is
 1200	        # offered now, as its own full event — stored, never dropped.
 1201	        if pending:
 1202	            try:
 1203	                while await deliver_inbox_one() is not None:
 1204	                    pass
 1205	            except _NoDevices:
 1206	                pass  # the band vanished mid-delivery; entries stay pending
 1207	        # Keep the socket alive and drain client frames. The dev band never
 1208	        # sends server-meaningful frames yet; input arrives over HTTP (the
 1209	        # Omni spike's shape).
 1210	        while True:
 1211	            await ws.receive_text()
 1212	    except WebSocketDisconnect:
 1213	        pass
 1214	    finally:
 1215	        hub.detach(ws)

 1235	@app.post("/inject/audio")
 1236	async def inject_audio(
 1237	    body: dict[str, Any],
 1238	    authorization: str | None = Header(default=None),
 1239	) -> JSONResponse:
 1240	    """The Omni spike's seam (week 1), now real: mic audio → transcript → turn.
 1241	
 1242	    The PWA records the mic (MediaRecorder) and posts
 1243	    ``{"audio_b64": "…", "mime": "webm", "lang": "en"}``. Without an
 1244	    audio-capable endpoint configured this answers 503 — a refusal, never a
 1245	    silent failure. ``{"simulate": "<text>"}`` is the keyless path the demo
 1246	    runs; it takes the *same turn* a typed message would, so cues, ticks,
 1247	    cells, close and the queue are all exercised either way.
 1248	    """
 1249	    _authorized_http(authorization)
 1250	    simulate = str(body.get("simulate") or "").strip()
 1251	    audio_b64 = str(body.get("audio_b64") or "").strip()
 1252	    if not simulate and not audio_b64:
 1253	        raise HTTPException(
 1254	            status_code=422,
 1255	            detail="provide 'audio_b64' (+ 'mime': webm|ogg|mp4|mp3|wav) "
 1256	            "or 'simulate' (keyless stub)",
 1257	        )
 1258	    if simulate:
 1259	        text, source = simulate, "audio_stub"
 1260	    else:
 1261	        if not NEBIUS_API_KEY:
 1262	            raise HTTPException(
 1263	                status_code=503,
 1264	                detail="NEBIUS_API_KEY is not set — transcription unavailable; "
 1265	                "use {\"simulate\": \"…\"} for the keyless path",
 1266	            )
 1267	        if not ZIV_OMNI_MODEL:
 1268	            raise HTTPException(
 1269	                status_code=503,
 1270	                detail="no audio-capable endpoint configured — set "
 1271	                "ZIV_OMNI_MODEL to an endpoint that accepts audio input; "
 1272	                "use {\"simulate\": \"…\"} for the keyless path",
 1273	            )
 1274	        mime = str(body.get("mime") or "webm").strip().lower()
 1275	        allowed = {"webm", "ogg", "mp4", "mp3", "wav"}
 1276	        if mime not in allowed:
 1277	            raise HTTPException(
 1278	                status_code=422,
 1279	                detail=f"unsupported mime {mime!r} — one of {sorted(allowed)}",
 1280	            )
 1281	        try:
 1282	            text = await _transcribe_omni(audio_b64, mime)
 1283	        except HTTPException as exc:
 1284	            # The error long-buzz, not silence (the spike's loud-not-silent
 1285	            # rule): the transcription died before a turn could start, so
 1286	            # the wrist feels the attention beat now.
 1287	            await _announce_error(f"transcription failed ({exc.status_code})")
 1288	            raise
 1289	        source = "audio_omni"
 1290	    try:
 1291	        result = await run_message_turn(text, source=source)
 1292	    except _NoDevices as exc:
 1293	        raise HTTPException(status_code=409, detail=str(exc)) from exc
 1294	    return JSONResponse({"source": source, **result})

 1358	@app.get("/api/ziv/health")
 1359	def health() -> dict[str, Any]:
 1360	    """Operator view — deliberately public, and deliberately content-free.
 1361	
 1362	    The PWA's queue badge polls this before it has any credential, so it is
 1363	    not bearer-gated. It answers counts, caps, model wiring, turn telemetry
 1364	    and corrupt-store flags — never a message's text, a reminder's line, or
 1365	    a preference value. Anything added here must keep that property; the
 1366	    routes that DO carry content are the ``AUTHENTICATED`` ones.
 1367	    """
 1368	    # Every load that can DISCOVER corruption runs BEFORE the single drain:
 1369	    # a corrupt file is found (and reported) by the depth/count/prefs reads,
 1370	    # so the drain below captures all of it — one snapshot, the flag and
 1371	    # the store_problems list can never disagree, and the consume-once
 1372	    # report is consumed exactly once.
 1373	    schedule_pending = scheduled.count()
 1374	    inbox_pending = inbox.count()
 1375	    prefs = {"cell_gap_ms": _effective_cell_gap_ms()}
 1376	    corrupt = (
 1377	        memory.take_corrupt_files()
 1378	        + inbox.take_corrupt_files()
 1379	        + scheduled.take_corrupt_files()
 1380	    )
 1381	    return {
 1382	        "status": "ok",
 1383	        "service": "Ziv relay v1 (dev band + memory)",
 1384	        "devices": hub.count,
 1385	        "spec_version": timing.SPEC_VERSION,
 1386	        "fake_model_seconds": FAKE_MODEL_SECONDS,
 1387	        "auth": bool(ZIV_RELAY_TOKEN),
 1388	        # The model wiring, operator-visible: which path can actually run
 1389	        # right now. `text` is the demo's model; `omni` is the audio seam,
 1390	        # unset unless an audio-capable endpoint is configured.
 1391	        "text": {
 1392	            "key_set": bool(NEBIUS_API_KEY),
 1393	            "base_url": NEBIUS_BASE_URL.rstrip("/"),
 1394	            "model": ZIV_TEXT_MODEL,
 1395	            "available": agent_path_available(),
 1396	        },
 1397	        "omni": {
 1398	            "key_set": _omni_configured(),
 1399	            "model": ZIV_OMNI_MODEL,
 1400	            "configured": bool(ZIV_OMNI_MODEL),
 1401	        },
 1402	        "gate_queue": len(gate),
 1403	        "gate_queue_cap": MessageGate.MAX_QUEUED,
 1404	        "queue_rejections": queue_rejections.snapshot(),
 1405	        "inbox_pending": inbox_pending,
 1406	        # Cap pressure, made visible. The inbox and the schedule still drop
 1407	        # their oldest entry when full — that is what a cap means — but the
